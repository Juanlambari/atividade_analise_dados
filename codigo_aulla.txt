int triger = 3;
int echo = 2;
int duracao = 0;
int distancia = 0;
void setup()
{
  Serial.begin(9600);
  pinMode(triger, OUTPUT);
  pinMode(echo, INPUT);
  pinMode(7, OUTPUT);
  pinMode(6, OUTPUT);
}

void loop()
{
  int posi = map(distancia, 0, 1024, 20, 320);
  digitalWrite(triger, HIGH);
  delayMicroseconds(10); // Wait for 1000 millisecond(s)
  digitalWrite(triger, LOW);
  duracao = pulseIn(echo, HIGH);
  distancia = duracao * 0.017175;
  Serial.print("Distancia: ");
  Serial.println(posi);
  
  if (posi >= 20 && posi <= 30){
  	tone(10, 174);
    digitalWrite(6, HIGH);
  }else if(posi >= 35 && posi <= 45){
  	tone(10, 285);
    digitalWrite(5, HIGH);
  }else if(posi >= 55 && posi <= 65){
  	tone(10, 396);
    digitalWrite(4, HIGH);
  }
  else if(posi >= 70 && posi <= 80){
  	tone(10, 417);
    digitalWrite(8, HIGH);
  }
  else if(posi >= 85 && posi <= 95){
  	tone(10, 432);
    digitalWrite(9, HIGH);
  }
  else if(posi >= 100 && posi <= 105){
  	tone(10, 528);
    digitalWrite(7, HIGH);
  }else{
    digitalWrite(6, LOW);
    delay(500);
    digitalWrite(5, LOW);
    delay(500);
    digitalWrite(4, LOW);
    delay(500);
    digitalWrite(8, LOW);
    delay(500);
    digitalWrite(9, LOW);
    delay(500);
  	digitalWrite(7, LOW);
  }
  delay(150); // Wait for 1000 millisecond(s)
}
